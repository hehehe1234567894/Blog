{{- /* 覆盖 PaperMod 的封面局部模板：文章没写 cover.image 时，自动按标题生成一张 SVG 封面 （这样新写的文章不用手动配图，也不会出现“有卡片没图”的空档） */ -}} {{- with .cxt -}} {{- $isHidden := .Params.cover.hidden | default (cond $.IsSingle (.Params.cover.hiddenInSingle) (.Params.cover.hiddenInList)) | default false -}} {{- $img := .Params.cover.image -}} {{- $auto := "" -}} {{- if and (not $img) (not $isHidden) (not $.IsSingle) (eq .Section "posts") -}} {{- $title := .Title -}} {{- $tags := .Params.tags | default slice -}} {{- /* 按标题长度+标签数选一套配色 */ -}} {{- $pals := slice (slice "#0a2540" "#1d4ed8" "#67e8f9") (slice "#2a1040" "#7c3aed" "#f0abfc") (slice "#0d2b22" "#047857" "#6ee7b7") (slice "#3a1a10" "#c2410c" "#fdba74") (slice "#1b1b3a" "#4338ca" "#a5b4fc") (slice "#2b1030" "#be185d" "#f9a8d4") -}} {{- $i := mod (add (len $title) (mul 7 (len $tags))) 6 -}} {{- $pal := index $pals $i -}} {{- $chars := split $title "" -}} {{- $l1 := delimit (first 12 $chars) "" -}} {{- $l2 := "" -}} {{- $l3 := "" -}} {{- if gt (len $chars) 12 -}}{{- $l2 = delimit (first 12 (after 12 $chars)) "" -}}{{- end -}} {{- if gt (len $chars) 24 -}}{{- $l3 = delimit (first 12 (after 24 $chars)) "" -}}{{- end -}} {{- /* -}} {{- $head := index (split $cur "") 0 -}} {{- if in $punc $head -}} {{- if eq $seg "l2" -}} {{- $l1 = printf "%s%s" $l1 $head -}} {{- $l2 = delimit (after 1 (split $cur "")) "" -}} {{- else -}} {{- $l2 = printf "%s%s" $l2 $head -}} {{- $l3 = delimit (after 1 (split $cur "")) "" -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $sub := delimit $tags " · " -}} {{- if not $sub -}}{{- $sub = .Date.Format "2006-01-02" -}}{{- end -}} {{- $svg := printf `` (index $pal 0) (index $pal 1) (index $pal 2) (index $pal 2) (index $pal 2) $l1 $l2 $l3 $sub -}} {{- $name := printf "covers/auto-%s.svg" (.File.BaseFileName | default (urlize .Title)) -}} {{- with resources.FromString $name $svg -}} {{- $auto = .RelPermalink -}} {{- end -}} {{- end -}} {{- $src := $img | default $auto -}} {{- if $src -}} {{- end -}} {{- end -}}
